cmake_minimum_required(VERSION 3.24)
project(Gomoku VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()

option(GOMOKU_BUILD_WORKER "Build the JSON Lines adapter (downloads nlohmann/json)" ON)
option(GOMOKU_BUILD_GOMOCUP "Build the standalone competition adapter" ON)
option(GOMOKU_BUILD_WASM "Build the browser WebAssembly adapter" OFF)
option(GOMOKU_BUILD_TESTS "Build native tests" ON)
option(GOMOKU_STATIC_RUNTIME "Use a standalone Windows runtime" ON)

add_library(gomoku_options INTERFACE)
target_compile_options(gomoku_options INTERFACE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>
  $<$<CXX_COMPILER_ID:MSVC>:/W4;/utf-8>)
if(WIN32)
  target_compile_definitions(gomoku_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(EMSCRIPTEN)
  # Search uses exceptions to unwind make/unmake on time and node limits.
  target_compile_options(gomoku_options INTERFACE -fexceptions)
  target_link_options(gomoku_options INTERFACE -fexceptions)
endif()
if(GOMOKU_STATIC_RUNTIME)
  if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_link_options(gomoku_options INTERFACE -static -static-libgcc -static-libstdc++)
  endif()
endif()

add_subdirectory(engine)
if(GOMOKU_BUILD_GOMOCUP)
  add_subdirectory(apps/gomocup)
endif()
if(GOMOKU_BUILD_WORKER OR GOMOKU_BUILD_WASM)
  include(FetchContent)
  FetchContent_Declare(nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(nlohmann_json)
  add_subdirectory(adapters/json)
endif()
if(GOMOKU_BUILD_WORKER)
  add_subdirectory(apps/engine-worker)
endif()
if(GOMOKU_BUILD_WASM)
  add_subdirectory(adapters/wasm)
endif()
if(GOMOKU_BUILD_TESTS)
  enable_testing()
  add_subdirectory(engine/tests)
endif()

set(CPACK_GENERATOR ZIP)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_PACKAGE_FILE_NAME "gomoku-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
